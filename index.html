async function uploadFiles(newFiles) { let totalCurrentBytes = filesList.reduce((sum, f) => sum + f.size, 0); let incomingBytes = Array.from(newFiles).reduce((sum, f) => sum + f.size, 0); let maxBytes = userQuotaGB * 1024 * 1024 * 1024; if ((totalCurrentBytes + incomingBytes) > maxBytes) { return showToast('Xotira yetarli emas!'); } setLoading(true, 'Fayllar yuklanmoqda...'); let filesArray = Array.from(newFiles); let successCount = 0; for (let f of filesArray) { try { let fileObj = await new Promise((resolve, reject) => { let reader = new FileReader(); reader.onload = () => resolve({ id: 'f_' + Date.now() + Math.random().toString(36).substr(2, 6), name: f.name, size: f.size, type: f.type || 'application/octet-stream', data: reader.result }); reader.onerror = error => reject(error); reader.readAsDataURL(f); }); filesList.push(fileObj); successCount++; } catch (e) { console.error("Faylni o'qishda xatolik:", f.name); } } if (successCount > 0) { saveFilesLocally(); renderFiles(); showToast(`${successCount} ta fayl muvaffaqiyatli saqlandi!`); } else { showToast('Fayllarni yuklashda xatolik yuz berdi!'); } setLoading(false); }
